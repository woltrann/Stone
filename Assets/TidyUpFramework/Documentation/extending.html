Example: a custom before/after reactor

using TidyUpFramework.Transformation;
using UnityEngine;

public sealed class TidyAudioPitchReactor : TidyProgressReactor
{
    [SerializeField] AudioSource source;
    [SerializeField] float messyPitch = 0.85f;

    protected override void Apply(float organisedAmount)
    {
        source.pitch = Mathf.Lerp(messyPitch, 1f, organisedAmount);
    }
}

Example: a HUD counter for mistakes

using TidyUpFramework.Levels;
using TidyUpFramework.Placement;
using TidyUpFramework.UI;
using TMPro;
using UnityEngine;

public sealed class MistakeCounterView : TidyLevelBoundView
{
    [SerializeField] TMP_Text label;

    protected override void Bind(TidyLevelManager manager)
    {
        manager.PlacementEvaluated += Refresh;
    }

    protected override void Unbind(TidyLevelManager manager)
    {
        manager.PlacementEvaluated -= Refresh;
    }

    void Refresh(TidyPlacementEvent placement)
    {
        label.text = LevelManager.Context.Mistakes.ToString();
    }
}
